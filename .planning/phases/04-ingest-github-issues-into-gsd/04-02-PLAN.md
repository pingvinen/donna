Create `scripts/post-release-comments.cjs` following the `determine-bump.cjs` pattern: pure functions exported for testing, CLI entry point when run directly.

**File structure:**
```javascript
#!/usr/bin/env node
"use strict";

const fs = require("node:fs");
const path = require("node:path");
const { execSync } = require("node:child_process");
```

**Exported pure function: `scanDoneTodos(doneDir)`**
- Takes a directory path as argument
- Reads all `.md` files in the directory
- For each file, extracts `github_issue: <number>` from frontmatter using regex `/^github_issue:\s*(\d+)/m`
- Returns an object keyed by issue number, values are arrays of TODO file basenames: `{ 13: ["2026-03-26-add-ascii-art.md"], 21: [...] }`
- If directory doesn't exist or is empty, returns `{}`
- Gracefully skips files without `github_issue` field

**Exported pure function: `findMergedPRs(prevTag)`**
- Calls `gh pr list --repo pingvinen/donna --state merged --limit 100 --json number,mergedAt` via `execSync`
- Gets previous tag's commit date via `git log -1 --format=%aI ${prevTag}` via `execSync`
- Filters PRs to those merged AFTER the previous tag date
- Returns array of PR numbers: `[42, 43, 45]`
- If `prevTag` is empty/null or no PRs found, returns `[]`

**Helper function: `runGh(args)`**
- Wraps `execSync("gh " + args, { encoding: "utf8", stdio: ["pipe", "pipe", "pipe"] })`
- Catches errors and logs them but does not throw (graceful degradation per pitfall 6)

**CLI entry point (`if (require.main === module)`):**

Takes version as `process.argv[2]` (e.g., `0.9.0`). Exits with error message if not provided.

    console.log(`Issue #${issueNum} already closed, skipping`);

// Close with comment (D-11) — atomic comment+close per RESEARCH anti-patterns

**Step 3: Comment on merged PRs (per D-12)**
```javascript
const prevTag = execSync("git describe --abbrev=0 --tags HEAD^", { encoding: "utf8" }).trim();
const prNumbers = findMergedPRs(prevTag);
for (const prNum of prNumbers) {
    runGh(`pr comment ${prNum} --repo pingvinen/donna --body "Released in v${version}"`);
    console.log(`Commented on PR #${prNum}`);
}
```
If no previous tag exists (first release), skip PR commenting gracefully.

```javascript
"use strict";

const { describe, it, beforeEach, afterEach } = require("node:test");
const assert = require("node:assert/strict");
const fs = require("node:fs");
const path = require("node:path");
const os = require("node:os");

const { scanDoneTodos } = require("../scripts/post-release-comments.cjs");

describe("scanDoneTodos", () => {
    let tmpDir;

    beforeEach(() => {
        tmpDir = fs.mkdtempSync(path.join(os.tmpdir(), "donna-test-"));
    });

    afterEach(() => {
        fs.rmSync(tmpDir, { recursive: true, force: true });
    });

    it("returns empty object for non-existent directory", () => {
        const result = scanDoneTodos("/non/existent/path");
        assert.deepStrictEqual(result, {});
    });

    it("returns empty object for empty directory", () => {
        const result = scanDoneTodos(tmpDir);
        assert.deepStrictEqual(result, {});
    });

    it("extracts github_issue from frontmatter", () => {
        fs.writeFileSync(path.join(tmpDir, "test-todo.md"), [
            "---",
            "created: 2026-03-26T00:00:00.000Z",
            "title: Fix something (ref: #42)",
            "github_issue: 42",
            "files: []",
            "---",
            "",
            "## Problem",
            "Something is broken",
        ].join("\n"));

        const result = scanDoneTodos(tmpDir);
        assert.deepStrictEqual(result, { 42: ["test-todo.md"] });
    });

    it("groups multiple TODOs by issue number", () => {
        fs.writeFileSync(path.join(tmpDir, "todo-a.md"), "---\ngithub_issue: 10\n---\n");
        fs.writeFileSync(path.join(tmpDir, "todo-b.md"), "---\ngithub_issue: 10\n---\n");

        const result = scanDoneTodos(tmpDir);
        assert.strictEqual(result[10].length, 2);
    });

    it("skips files without github_issue field", () => {
        fs.writeFileSync(path.join(tmpDir, "no-issue.md"), "---\ntitle: Something\n---\n");

        const result = scanDoneTodos(tmpDir);
        assert.deepStrictEqual(result, {});
    });

    it("skips non-md files", () => {
        fs.writeFileSync(path.join(tmpDir, "notes.txt"), "github_issue: 99");

        const result = scanDoneTodos(tmpDir);
        assert.deepStrictEqual(result, {});
    });
});
```
